\section{Problem statement}

\paragraph{Sets and data.}
\[
\A=\{\text{brokers/agencies}\},\qquad
\C=\{\text{Gold},\,\text{Silver},\,\text{Bronze}\}.
\]
\[
W=\big[W_{a,c}\big]_{a\in\A,\;c\in\C}\ \text{(productivity weights)},\qquad
U=\big[U_a\big]_{a\in\A}\ \text{(per-agency capacity)}.
\]

\[
X \in \mathbb{Z}_{\ge 0}^{m\times p};\qquad
W \in \mathbb{R}_{\ge 0}^{m\times p}.
\]


\paragraph{Decision variables.}
\[
X=\big[X_{a,c}\big]_{a\in\A,\;c\in\C},\qquad
X_{a,c}\in \Z_{\ge 0}\ \ \text{(policies assigned to broker $a$ in bucket $c$)}.
\]

\paragraph{Objective (maximize total productivity).}
\noindent We use the Frobenius inner product $\langle W,X\rangle := \sum_{a\in\A}\sum_{c\in\C} W_{a,c}\,X_{a,c}$.
\begin{equation*}
\max\ \ \langle W,X\rangle
\;=\;
\sum_{a\in\A}\sum_{c\in\C} W_{a,c}\,X_{a,c}.
\end{equation*}

\paragraph{Constraints.}
\paragraph{Constraints in words (logical vs. business).}
\paragraph{Logical constraints.}
\begin{enumerate}[leftmargin=*]
\item \textbf{Integrality \& nonnegativity:} policies are whole and nonnegative,
$X_{a,c}\in\Z_{\ge 0}$.
\item \textbf{Global conservation:} total assigned equals the total number of policies
available to assign in that run (the sum of per-agency capacities),
$\mathbf{1}_A^\top X\,\mathbf{1}_C=U_{\mathrm{tot}}$ (or $U_{\mathrm{tot}}+1$ online).
\end{enumerate}

\paragraph{Business constraints.}
\begin{enumerate}[leftmargin=*]
\item \textbf{Per-agency capacity:} no agency exceeds its capacity,
$X\,\mathbf{1}_C \le U$.
\item \textbf{Previous-assignment floor (online):} do not drop an agency below its
batch total, $X\,\mathbf{1}_C \ge L$.
\item \textbf{ZIP admissibility (online):} if a ZIP is inadmissible for agency $a$,
lock its row total, $e_a^\top X\,\mathbf{1}_C=L_a$.
\item \textbf{Bucket mix:} keep each bucket total within bounds,
$L^{\text{bucket}} \le X^\top \mathbf{1}_A \le U^{\text{bucket}}$.
\end{enumerate}

\begin{align*}
\text{Per-broker capacity:}\quad
& \sum_{c\in\C} X_{a,c} \;\le\; U_a,
& \forall a\in\A
\\[4pt]
\text{Global conservation:}\quad
& \sum_{a\in\A}\sum_{c\in\C} X_{a,c} \;=\; U_{\mathrm{tot}}
\ \ (\text{or }U_{\mathrm{tot}}+1\ \text{in online mode}),
\\[4pt]
\text{Bucket mix (optional):}\quad
& L^{\text{bucket}}_c \ \le\ \sum_{a\in\A} X_{a,c}\ \le\ U^{\text{bucket}}_c,
& \forall c\in\C
\\[4pt]
\text{No decreases (online):}\quad
& \sum_{c\in\C} X_{a,c} \ \ge\ L^{\text{(batch)}}_a,
& \forall a\in\A
\\[4pt]
\text{ZIP lock (online, if inadmissible):}\quad
& \sum_{c\in\C} X_{a,c} \;=\; L^{\text{(Batch)}}_a,
& \forall a\in\A\ \text{inadmissible.}
\end{align*}

\section{Constraints (all in matrix form)}

Let $\mathbf{1}_A\in\R^{|\A|}$ and $\mathbf{1}_C\in\R^{|\C|}$ be all-ones vectors.

\begin{itemize}[leftmargin=*]

\item \textbf{Per-agency capacity.} Row sums are $X\,\mathbf{1}_C$. Require
\begin{equation*}
X\,\mathbf{1}_C \;\le\; U.
\end{equation*}

\item \textbf{Global conservation.} 

Total assigned equals inventory 
$U_{\mathrm{tot}}=\mathbf{1}_A^\top U$ (batch) or $U_{\mathrm{tot}}+1$ (online):
\begin{equation*}
\mathbf{1}_A^\top X\,\mathbf{1}_C \;=\; U_{\mathrm{tot}}
\quad\text{(batch)}
\qquad\text{or}\qquad
\mathbf{1}_A^\top X\,\mathbf{1}_C \;=\; U_{\mathrm{tot}} + 1
\quad\text{(online)}.
\end{equation*}

\item \textbf{Bucket mix (optional).} With bounds 
$L^{\text{bucket}},\,U^{\text{bucket}} \in \mathbb{R}^{|\C|}$ and column sums $X^\top \mathbf{1}_A$:
\begin{equation*}
L^{\text{bucket}} \;\le\; X^\top \mathbf{1}_A \;\le\; U^{\text{bucket}}.
\end{equation*}

\item \textbf{online floors (no decreases).} With $L$ the batch row totals,
\begin{equation*}
X\,\mathbf{1}_C \;\ge\; L.
\end{equation*}

\item \textbf{ZIP lock (online, inadmissible $a$).} With $e_a$ the $a$-th basis vector,
\begin{equation*}
e_a^\top X\,\mathbf{1}_C \;=\; L_a.
\end{equation*}

\item \textbf{Online headroom (feasibility only).} Optionally allow one extra slot
per agency that still had capacity:
\begin{equation*}
X\,\mathbf{1}_C \;\le\; U + h,
\qquad
h_a=\begin{cases}
1, & U_a>0 \ \text{and online},\\
0, & \text{otherwise}.
\end{cases}
\end{equation*}

\item \textbf{Integrality \& nonnegativity.}
\begin{equation*}
X \in \mathbb{Z}_{\ge 0}^{\,|\A|\times |\C|}.
\end{equation*}

\end{itemize}

\section{Compact vectorized (Kronecker) form (optional)}

Vectorize column-wise: $x=\vecop(X)\in\Z_{\ge 0}^{|\A||\C|}$, $w=\vecop(W)$.
Define the summation operators
\[
R=(\onesC^\top\!\otimes I_{|\A|})\in \R^{|\A|\times |\A||\C|}\quad\text{(row sums)},\qquad
C=(I_{|\C|}\!\otimes \onesA^\top)\in \R^{|\C|\times |\A||\C|}\quad\text{(col sums)},
\]
\[
G=(\onesC^\top\!\otimes \onesA^\top)\in \R^{1\times |\A||\C|}\quad\text{(global sum)}.
\]
Then the MILP is
\begin{equation*}
\begin{aligned}
\max\ & w^\top x \\
\text{s.t. }\
& R x \le U,\\
& C x \in \big[L^{\text{bucket}},\, U^{\text{bucket}}\big],\\
& G x = U_{\mathrm{tot}} \ \ (\text{or }U_{\mathrm{tot}}+1\ \text{online}),\\
& -R x \le -L^{\text{(batch)}}\ \ (\text{online}),\\
& (R_{a,\cdot})x=L^{\text{(batch)}}_a\ \ \text{for inadmissible }a\ \ (\text{online}),\\
& x\in \Z_{\ge 0}^{|\A||\C|}.
\end{aligned}
\end{equation*}

\section{Alternative objective: maximum entropy (optional)}
If you want a neutral allocation (spread policies as evenly as possible) rather than
maximizing productivity, you can replace the objective with a maximum-entropy criterion.
Let $x_{a,c}\ge 0$ be \emph{continuous} (relaxing integrality). The entropy objective is
\[
\max\ \ -\sum_{a\in\A}\sum_{c\in\C} x_{a,c}\log x_{a,c}
\quad\text{(equivalently, maximize } \sum_{a,c}\mathrm{entr}(x_{a,c})\text{)}.
\]
Subject to the same linear constraints (capacity, conservation, bucket mix, online floors,
ZIP lock), this yields the most ``uniform'' feasible assignment. Because the entropy
objective is nonlinear, this is a convex program (not an ILP); you can solve it with
convex solvers or approximate it via piecewise-linear constraints if you need integrality.

\section{Hybrid objective: productivity + entropy (optional)}
When you trust historical weights but want neutrality where information is weak, add
an entropy regularizer:
\[
\max\ \ \sum_{a\in\A}\sum_{c\in\C} W_{a,c}x_{a,c}\;+\;\lambda\sum_{a\in\A}\sum_{c\in\C}\mathrm{entr}(x_{a,c}),
\qquad \lambda>0.
\]
This keeps the productivity objective dominant while encouraging a smoother spread
of assignments when the weights are similar or uncertain. As with pure entropy,
this is a convex (continuous) problem; for integrality you can use a piecewise-linear
approximation or solve the continuous relaxation and then round/repair.

\section*{Toy data (4 agencies $\times$ 3 buckets)}
Agencies $\A=\{\mathrm{A1},\mathrm{A2},\mathrm{A3},\mathrm{A4}\}$ and buckets
$\C=\{\mathrm{Gold},\mathrm{Silver},\mathrm{Bronze}\}$. Decision matrix
$X\in\mathbb{Z}_{\ge 0}^{4\times 3}$; objective is $\max\langle W,X\rangle$.

Per-agency capacities and totals:
\[
U=\begin{bmatrix}4\\[2pt]3\\[2pt]1\\[2pt]2\end{bmatrix},
\quad
U_{\mathrm{tot}}=10.
\]

Bucket bounds:
\[
L^{\text{bucket}}=\begin{bmatrix}2\\[2pt]3\\[2pt]2\end{bmatrix},\quad
U^{\text{bucket}}=\begin{bmatrix}7\\[2pt]7\\[2pt]6\end{bmatrix}.
\]

Productivity weights (bucket-adjusted):
\[
W=
\begin{array}{c|ccc}
 & \text{Gold} & \text{Silver} & \text{Bronze} \\\hline
\text{A1} & 12.03 & 10.00 & 8.51 \\
\text{A2} & 8.41 & 7.04 & 5.98 \\
\text{A3} & 6.04 & 5.00 & 4.27 \\
\text{A4} & 9.60 & 8.01 & 6.83
\end{array}
\]

Online ZIP admissibility (example): incoming ZIP is admissible for A1 and A2, but
inadmissible for A3 and A4, so their row totals are locked at the online floor.

\section*{Batch optimization (no new policy)}
\[
\begin{aligned}
\text{maximize}\quad & \langle W,X\rangle \\[2pt]
\text{subject to}\quad
& X\,\mathbf{1}_C \le U, \\[2pt]
& \mathbf{1}_A^\top X\,\mathbf{1}_C = U_{\text{tot}}=10, \\[2pt]
& L^{\text{bucket}} \le X^\top \mathbf{1}_A \le U^{\text{bucket}}, \\[2pt]
& X \in \mathbb{Z}_{\ge 0}^{4\times 3}.
\end{aligned}
\]

An optimal batch solution is
\[
X^{(0)}=
\begin{array}{c|ccc}
 & \text{Gold} & \text{Silver} & \text{Bronze} \\\hline
\text{A1} & 4 & 0 & 0 \\
\text{A2} & 0 & 2 & 1 \\
\text{A3} & 0 & 0 & 1 \\
\text{A4} & 1 & 1 & 0
\end{array}
\]
with row totals $\begin{bmatrix}4\\[2pt]3\\[2pt]1\\[2pt]2\end{bmatrix}$ and
objective $\langle W,X^{(0)}\rangle=90.069$.

\section*{Online optimization (+1 policy) with ZIP locks}
Online rules:
\begin{itemize}[leftmargin=*]
\item \textbf{Global:} $\mathbf{1}_A^\top X\mathbf{1}_C=11$ (exactly one more).
\item \textbf{Floors (no decreases):} $X\mathbf{1}_C \ge L$, where $L:=X^{(0)}\mathbf{1}_C$.
\item \textbf{Headroom (feasibility only):} allow one extra slot per agency that had capacity:
$X\mathbf{1}_C \le U+h$, with $h_a=1$ if $U_a>0$ and online, else $0$.
\item \textbf{ZIP locks:} incoming ZIP is inadmissible for A3 and A4, so
$e_3^\top X\mathbf{1}_C=L_3$ and $e_4^\top X\mathbf{1}_C=L_4$.
\item \textbf{Bucket bounds:} keep $L^{\text{bucket}}\le X^\top \mathbf{1}_A \le U^{\text{bucket}}$.
\end{itemize}

An optimal online solution is
\[
X^{(1)}=
\begin{array}{c|ccc}
 & \text{Gold} & \text{Silver} & \text{Bronze} \\\hline
\text{A1} & 5 & 0 & 0 \\
\text{A2} & 0 & 2 & 1 \\
\text{A3} & 0 & 0 & 1 \\
\text{A4} & 1 & 1 & 0
\end{array}
\]
with objective $\langle W,X^{(1)}\rangle=102.101$ and delta
$X^{(1)}\mathbf{1}_C-X^{(0)}\mathbf{1}_C=[1,0,0,0]^\top$.
